cmake_minimum_required(VERSION 3.20)
project(kernel_apc_test C)
if(NOT WIN32)
    message(FATAL_ERROR "This synthetic-memory test uses Windows VirtualAlloc and a host temp file")
endif()
set(CMAKE_C_STANDARD 11)
add_subdirectory(../.. runtime)
add_executable(kernel_apc_test test_main.c)
target_include_directories(kernel_apc_test PRIVATE ../../src/kernel)
target_link_libraries(kernel_apc_test PRIVATE xboxrecomp)
enable_testing()
# RECOMP_APC_DEFERRED is read once per process: one run per mode.
add_test(NAME kernel_apc_inline COMMAND kernel_apc_test)
add_test(NAME kernel_apc_deferred COMMAND kernel_apc_test deferred)
