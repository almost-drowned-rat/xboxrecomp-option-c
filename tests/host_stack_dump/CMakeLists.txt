cmake_minimum_required(VERSION 3.20)
project(host_stack_dump_test C)
set(CMAKE_C_STANDARD 11)
# src/platform/host_stack_dump.c alone: no runtime, no SDL.
add_executable(host_stack_dump_test test_main.c ../../src/platform/host_stack_dump.c)
target_include_directories(host_stack_dump_test PRIVATE ../../src)
if(NOT MSVC)
    target_compile_options(host_stack_dump_test PRIVATE -Wall -Wextra -fno-omit-frame-pointer)
endif()
if(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(host_stack_dump_test PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
endif()
enable_testing()
add_test(NAME host_stack_dump_test COMMAND host_stack_dump_test)
