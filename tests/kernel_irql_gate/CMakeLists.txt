cmake_minimum_required(VERSION 3.20)
project(kernel_irql_gate_test C)
if(NOT WIN32)
    message(FATAL_ERROR "This test uses Windows threads and events")
endif()
set(CMAKE_C_STANDARD 11)
add_subdirectory(../.. runtime)
add_executable(kernel_irql_gate_test test_main.c)
target_include_directories(kernel_irql_gate_test PRIVATE ../../src/kernel)
target_link_libraries(kernel_irql_gate_test PRIVATE xboxrecomp)
enable_testing()
# Two arms: RECOMP_DPC_GATE is read once per process.
add_test(NAME kernel_irql_gate_on COMMAND kernel_irql_gate_test)
set_tests_properties(kernel_irql_gate_on PROPERTIES ENVIRONMENT "GATE_EXPECT_ON=1")
add_test(NAME kernel_irql_gate_off COMMAND kernel_irql_gate_test)
set_tests_properties(kernel_irql_gate_off PROPERTIES ENVIRONMENT "RECOMP_DPC_GATE=0;GATE_EXPECT_ON=0")
